  (function() {
      var hash = window.location.hash;
      if (hash && (
          hash.indexOf('#search/') === 0 ||
          hash.indexOf('#meeting/') === 0 ||
          hash.indexOf('#taba/') === 0 ||
          hash.indexOf('#request/') === 0 ||
          hash.indexOf('#building/') === 0 ||
          hash.indexOf('#gush/') === 0 ||
          hash.indexOf('#gushmakor/') === 0 ||
          hash.indexOf('#unified/') === 0
      )) {
          window._savedBackboneHash = hash;
          history.replaceState(null, '', window.location.pathname + window.location.search);

          function restoreHash() {
              setTimeout(function() {
                  if (window._savedBackboneHash) {
                      history.replaceState(null, '', window.location.pathname + window.location.search + window._savedBackboneHash);
                      if (window.Backbone && Backbone.history) {
                          Backbone.history.loadUrl(window._savedBackboneHash.substring(1));
                      }
                  }
              }, 100);
          }

          // If DOMContentLoaded already fired (e.g. on Next.js sites where scripts
          // are injected post-hydration), restore immediately instead of waiting
          if (document.readyState === 'loading') {
              document.addEventListener('DOMContentLoaded', restoreHash);
          } else {
              restoreHash();
          }
      }
  })();

// === תיקון תאריכון: בחירת שנה+חודש ולחיצה על האייקון (PROD/WordPress) ===
(function () {
  function fixAll() {
    if (!window.jQuery || !jQuery.fn.datepicker) return;
    jQuery('input.hasDatepicker').each(function () {
      var $i = jQuery(this);
      try {
        if ($i.datepicker('option', 'changeYear') !== true) {
          $i.datepicker('option', {
            changeMonth: true,
            changeYear: true,
            yearRange: 'c-100:c+1',
            dateFormat: 'dd/mm/yy',
            monthNames: ['ינואר','פברואר','מרץ','אפריל','מאי','יוני','יולי','אוגוסט','ספטמבר','אוקטובר','נובמבר','דצמבר'],
            monthNamesShort: ['ינואר','פברואר','מרץ','אפריל','מאי','יוני','יולי','אוגוסט','ספטמבר','אוקטובר','נובמבר','דצמבר']
          });
        }
      } catch (e) {}
    });
  }
  // רץ באופן עצמאי, לא תלוי ב-document.ready, ובודק שוב ושוב עד שהשדות מוכנים
  var tries = 0;
  var iv = setInterval(function () {
    fixAll();
    tries++;
    if (tries > 40) clearInterval(iv); // עד ~20 שניות, אז עוצר
  }, 500);
  // לחיצה על אייקון לוח השנה — בשלב ה-capture כדי לתפוס לפני jQuery UI
  document.addEventListener('click', function (e) {
    var icon = e.target.closest && e.target.closest('.glyphicon-calendar.circle-icon');
    if (!icon) return;
    e.preventDefault();
    var input = icon.parentNode.querySelector('input.form-control');
    if (input) {
      try { jQuery(input).datepicker('show'); }
      catch (err) { input.focus(); }
    }
  }, true);
})();

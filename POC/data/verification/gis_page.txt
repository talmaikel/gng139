    function drawCircleFunc() {
        var checkBox = document.querySelector("#checkboxRadius");
        var numberRadius = document.querySelectorAll(".numberRadius");
        if (checkBox.checked) {
            for (var i = 0; i < numberRadius.length; i++) {
                numberRadius[i].style.display = "initial";
                $sys.globals3.myNumberRadius = true;
            }
        } else {
            for (var i = 0; i < numberRadius.length; i++) {
                numberRadius[i].style.display = "none";
                $sys.globals3.myNumberRadius = false;
            }
        }
    }
